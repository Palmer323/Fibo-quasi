#  The following function delivers the set of Fibonacci
#  cycles (F-cycles) derived from a table or matrix  
#  (Latin Square) related to a quasigroup
 
#  fCycles(tab) returns a list of the F-cycles for the 
#  mutiplication table tab.

fCycles := function(tab)
	local used, seq, seqlist, n ,nextpair, formseq;
	n :=  DimensionsMat(tab)[1]; # GAP function delivering the order
	used :=  NullMat(n,n); # GAP function delivering a null matrix
	seq := [1];
	seqlist := [];
 
# nextpair is a local function which delivers the starting pair for the next
# F-cycle
	nextpair :=  function()
		local j , k, s1 , s2; 
		
		s1 := 0;
		s2 := 0;
		for j in [1..n] do
            for k in [1..n] do
			if used [j][k] = 0 then 
				s1 := j ;
				s2 := k ;
			fi;
	if s1 <> 0 then break; fi;
	od; 
	if s1 <> 0 then break; fi;
	od;
	return [s1,s2];
	end;

# formseq function forms the next F-cycle for multiplication table 
# tab. Entries in 'used' are set to 1 when pair is placed in the sequence. 

	formseq := function()	
		local next, current, last , seqq ;
		seqq := nextpair(); # obtain the first two elements of the sequence
		last := seqq[1];
		current := seqq[2];
		if last = 0 then return seqq ; fi;
		next := tab [last][current] ;
		while used [last][current] = 0 
		do
			used [last][current] := 1 ;	
			next := tab [last][current]  ;
			Add(seqq, next) ;	
			last := current ;
			current := next ;
		od ;

# remove the last two members which repeat the first 
	Remove(seqq);
	Remove(seqq);
	return seqq;
	end;

# main program
	while seq[1] > 0 	
	do
		seq := formseq(); 
		Add(seqlist,seq);
	od;
	Remove(seqlist);
	return seqlist;
end;


#  pSpectrum(tab) -- returns the P-spectrum; a list which 
#  contains the periods of each of the F-cycles related 
#  to the table tab (and generated by function fCycles). 
#  Returned in nin-decreasing order.  
#  The spectrum is a partition of the square of the order  
#  of the related quasigroup.

pSpectrum := function(tab)
local flist, n , partlist, el ;
partlist := [];
flist := fCycles(tab);
for el in flist do
	Add(partlist,Length(el));
	od;
Sort(partlist);
return partlist;
end;

end;

# shiftL is a function which shifts circularly a list one place left.

shiftL := function(list)
local temp;
temp := list[1];
Remove(list, 1);
Add(list, temp);
end;
 
# function to convert a set of fCycles to the corresponding to a related table
# fCyclesToTab(fcycles)
# delivers: the table for which generates a given list of F-cycles  
# code assumes fcycles is a legitimate representation of a Cayley table.
        
fCyclesToTab := function(fcycles)
local fseq,dim, dimsqd, tab,k,l;

# calculate order (dim) of quasigroup from fcycles 
dimsqd := 0;
for fseq in fcycles do
    dimsqd := dimsqd + Length(fseq);
od;
dim := RootInt(dimsqd);
if dim^2<>dimsqd then Print( "dimension error");return fail; fi;

tab :=  NullMat(dim ,dim); # initialise table
for fseq in fcycles do
    l := Length(fseq);
    for k in [1..l] do
        if l>2 then tab[fseq[1]][fseq[2]] := fseq[3]; fi;
        if l=2 then tab[fseq[1]][fseq[2]] := fseq[1]; fi;
        if l=1 then tab[fseq[1]][fseq[1]] := fseq[1]; fi;
        shiftL(fseq);
    od;
od; 
return tab;
end;

# reverseFcycles returns a set of cycles with each cycle reversed

reverseFcycles:=function(fcycles)
local cycles, e, r;
cycles:=[];
for e in fcycles do r:=Reversed(e); Add(cycles,r); od;
return cycles;
end;

# check if a sub spectrum (sub) is a sub-sequence of a given spectrum (main)

isSubSpectrum := function(main, sub)
local subPointer, mainPointer;
# both main and sub are assumed to be in increasing order (not checked)   
    subPointer := 1;
    mainPointer := 1;
    
    # If sub is empty, it is technically a subsequence
    if Length(sub) = 0 then return true; fi;
    
    # Iterate through the main list
    while mainPointer <= Length(main) do
        # If elements match, move the pointer for sub
        if main[mainPointer] = sub[subPointer] then
            subPointer := subPointer + 1;
        fi;
        
        # If all elements of sub are found
        if subPointer > Length(sub) then
            return true;
        fi;
        
        mainPointer := mainPointer + 1;
    od;
    
    # If loop finishes without matching all elements
    return false;
end;


# findSpectrumRand(dim,spectrum,repets) - generates random qgs until it finds # # the requested spectrum as a sublist of a spectrum of a generated qg.
# returns the multiplication table of the found quasigroup. 
# uses GAP standard function to obtain random Latin Square

findSpectrumRand:= function(dim,spectrum,repets)
local j, tab;
tab := [];
for j in [1..repets] do;
	tab := MultiplicationTable( RandomQuasigroup( dim) );

	if  isSubSpectrum( pSpectrum(tab), spectrum )  <> false 
		then return tab; fi;
od;
return  "Not found";
end;



# rotateToMin is a function which finds the minimum value of all the possible
# rotations of an Fcycle.
 
rotateToMin:= function(fcyc)
local mintodate,n,cy,ord,shifts;
shifts := Length(fcyc);  # number of times Fcycle will need to be shifted
ord:=Maximum(fcyc);
mintodate := []; # minimum found so far
cy := ShallowCopy(fcyc);
# calculate mintodate
for n in [1..shifts] do Add(mintodate,ord); od; # set up largest possible
for n in [1..shifts] do
    if cy < mintodate then mintodate:=ShallowCopy(cy); fi; 
    shiftL(cy);
od;
return mintodate;
end;




# function to verify if a given permutation provides an isomorphism
# between two quasigroups. Each Quasigroup is represented by two lists
# of F-cycles (cys1 and cys2). Perm is in d images of the integers 1,...,d 

verifyIso:=function(cys1,cys2,perm)
local transcys1,image1,image2,cy,e,cp,f;
transcys1:=[]; # cycles from applying perm to cys1 
image1:=[];
image2:=[];
if 0 in perm then return false; fi;  # this is partial permutation
for e in cys1 do   # apply the permutation to cys1 to get transcyc1
    cp:=[]; 
    for f in e do 
        Add(cp,perm[f]); 
    od;
    Add(transcys1,cp);
od;
# rotate all F-cycles to minimum to allow comparison
for e in transcys1 do AddSet(image1,rotateToMin(e));od;
for e in cys2 do AddSet(image2,rotateToMin(e)); od;
if image1=image2 then return true; fi;
return false; 
end;

# isomorphicFcycles Create an list of F-Cycles isomorphic to a given list
# by applying a given permutation. Returns the list of of F-cycles

isomorphicFcycles:= function(cycles,perm)
local e, cp, new, f;
new :=[];
for e in cycles do
   cp :=[]; for f in e do Add(cp,f^perm); od;
   Add(new,cp);
od;
return new;
end; 

# create a Cayley table isomorphic to a given table using a random
# permutation. Use this to create isomorphic pairs in test files

createIso:= function(tab)
local cy,dim,perm,p,isot,cyiso;
cy := fCycles(tab);
dim := DimensionsMat(tab)[1];
perm := [1..dim];
Shuffle(perm);
p := PermList( perm );
cyiso := isomorphicFcycles(cy,p);
isot:=fCyclesToTab(cyiso);
return isot;
end;




# MAIN ISOMORPHISM TEST PROGRAM

# isoTest(t1,t2,soln)
# function uses f-cycle representation to test for isomorphism of
# two quasigroups represented by Cayley tables.
# if soln = “M” returns list of the multiple possible permutations.
# if soln = ”S” returns first solution found.
# in both cases returns fail if no solution found.

isoTest:=function(t1,t2,soln)
local cys1, cys2, lns, e, combineIfCompat, perminit, dim, alignedPermSet, createPerm, m, compatperms, numcys, pos, setf1, setf2, inpperms, outperms, lcy,
solutions;

# local function to combine two partial permutations if they are compatible
#
# returns fail if not compatible
# returns partial or full permutation if compatible

combineIfCompat:=function(p1,p2)
local n,pout,ptemp,e;
pout:=[]; for n in [1..Length(p1)] do Add(pout,0); od; 
for n in [1..Length(p1)] do 
    if p2[n]=p1[n] then pout[n]:=p1[n]; fi;
    
    if p1[n]=0 and p2[n]>0 then pout[n]:=p2[n]; fi;

    if p2[n]=0 and p1[n]>0 then pout[n]:=p1[n]; fi;

    if p1[n]>0 and p2[n]>0 and p2[n]<>p1[n] then return fail; fi;
od;
 ptemp:=[]; for e in pout do if e<>0 then Add(ptemp,e); fi; od;
if IsDuplicateFree(ptemp) then return pout; fi;
return fail;
end;

# local function to create permutation or partial permutation which maps 
# between two F-cycles c1 and c2. 
# Returns fail if partial permutation does not exist

createPerm:= function(c1,c2)
local n,permx;
permx:=ShallowCopy(perminit);
for n in [1..Length(c1)] do 
    if permx[c1[n]] = 0 then if c2[n] in permx then return fail; fi;
    permx[c1[n]] := c2[n];fi;
od;
return permx;
end;

# local function to return the list of all partial permutations which map
# F-cycle cy1 to cy2. Returns empty list if noe exist
  
alignedPermSet := function(cy1,cy2)
local c1,c2,ln,m,p,perms,permr;
if Length(cy1)<>Length(cy2) then return fail; fi;
perms:=[];
c1:=ShallowCopy(cy1);
c2:=ShallowCopy(cy2);
ln:=Length(c1);
# rotate c2 through each position relative to c1 
for m in [1..ln] do
   permr := createPerm(c1,c2); # seek a permutation for current position
   if permr <> fail then Add(perms,permr); fi;
   shiftL(c2);  # to next position
od;
return perms;
end;


# local function receives two equal size lists of fcycles where all fcycles are
# the same length together with a set of permutations. 
# it returns a set of all permutations where each permutation is compatible 
# with at least one input permutation. ('permutations' may be partial)


compatperms := function(fcyc1,fcyc2,inpperms)
local outperms,ln,p,lp,ind,pset,e,f,newp,inputperms,failcount,reps; 

if Length(fcyc1)<>Length(fcyc2) then   Print(" Length error "); return fail; fi;

inputperms := ShallowCopy(inpperms);
outperms:=ShallowCopy(inpperms);

ln := Length(fcyc1); # number of equal sized F-cycles in fcyc1
reps := 0;
failcount:=0;
# All possible pairings of the cycles in cycl1 with cycles in cycl2 need to be
# considered. 
for p in SymmetricGroup(ln) do # iterate through all permutations of order ln
# this is very time consuming for the "M" option if ln is greater than 5 I. e.
# if the Pspectrum has more that 5 equal elements. Use only "S" option.
    lp := ListPerm(p,ln);
    # each p defines a pairing of the fcycles in fcyc1 to fcyc2
    for ind in [1..ln] do
       # obtain the permutations for the next pair
       pset := alignedPermSet(fcyc1[ind],fcyc2[lp[ind]]);
       if pset=[] then break; fi;
       # check each permutation in pset against each input permutat1;ion
       for e in pset do
           for f in inputperms do
               reps:=reps+1;
               newp := combineIfCompat(e,f);
               if newp <> fail then Add(outperms,newp);
               if verifyIso(cys1,cys2,newp) and soln="S" # solution for “S” case
               then return [newp]; fi;  fi;               
               if newp = fail then failcount:=failcount+1; fi;                                                                 
           od;       
       od;

if outperms=[] then outperms:= ShallowCopy(inputperms); fi;       
       outperms := DuplicateFreeList(outperms);
       inputperms:=ShallowCopy(outperms);         
    od;
od;
if failcount=reps then return []; fi; # failed every pairing
return outperms;    
end;


# initialisation
# clearly not isomorphic if spectra different
if pSpectrum(t1)<>pSpectrum(t2) then return fail; fi;

# obtain fCycles for t1 and sort in increasing order of length 
cys1 := fCycles(t1);
lns:=[]; for e in cys1 do Add(lns,Length(e));od;
SortParallel(lns,cys1);

# obtain fCycles for t2 and sort in increasing order of length
cys2 := fCycles(t2);
lns:=[]; for e in cys2 do Add(lns,Length(e));od;
SortParallel(lns,cys2);

# create null (partial) permutation to use at start of calculations
# permutations are stored as a list of the d images of the integers 1,...,d  
perminit := [];
dim := DimensionsMat(t1)[1];
for m in [1..dim] do Add(perminit,0); od;

numcys := Length(cys1);  # number of f-cycles for each table
inpperms := [ShallowCopy(perminit)];
outperms:=[];
solutions:=[];
pos := numcys; # set pointer to first of longest F-cycles
# process each 
repeat    
    setf1 := []; # to hold next list of equal length cycles from cys1
    setf2 := []; # to hold next list of equal length cycles from cys2
    lcy := Length(cys1[pos]);  # obtain length of cycles to be considered 
    while Length(cys1[pos])=lcy do # collect all cycles of this length 
        Add(setf1,cys1[pos]);
        Add(setf2,cys2[pos]);
        pos:=pos-1;
        if pos=0 then break; fi; # exhausted all cycles
    od;

    # call major local function to process the two sets of equal length cycles
    outperms := compatperms(setf1,setf2,inpperms);
    if outperms=[] then return fail; fi; # no solution for cycles of this
                                         # length. Therefor overall fail.

    for e in outperms do # determine any of returned potential permutations 
                     # provide a solution
        if verifyIso(cys1,cys2,e)and soln="S" then return PermList(e); fi;
        # solution found and only one sought so return
        if verifyIso(cys1,cys2,e)and soln="M"
             then AddSet(solutions,PermList(e)); fi;
        # solution found and multiple solutions sought – add to set of solutions
    od;

    inpperms := ShallowCopy(outperms);

until pos=0; 
if solutions=[] then return fail; fi;
if soln="S" then return fail; fi;
if soln="M" then return solutions; fi;
end;


# function which uses the standard LOOPS test for isomorphism of two tables.

isomorphtab := function(t1,t2)
local q1,q2,qp,p,ret;

   q1 := QuasigroupByCayleyTable(t1);
   q2 := QuasigroupByCayleyTable(t2);
   ret := IsomorphismQuasigroups(q1,q2);    
return ret;
end;

# createIsoTestList(dim,num) Test file creation for isomorphism test.
# Creates a list in which a random isomorphic Cayley table is inserted after 
# each randomly selected table.
# dim: the order of the tables to be generated
# num: the number of pairs to be created

createIsoTestList := function(dim,num)
local tr, t, n, isotestlist;

    Add(isotestlist,t);
    Add(isotestlist,createIso(t));

end;

# test1(tlist,soln) prints out the results of isomorphism test using list 
# created by createIsoTestList followed by standard LOOPS test result.

test1 := function(tlist,soln)
local n;
for n in [1..Length(tlist)-1] do 
    Print(n," isoTest ",isoTest(tlist[n],tlist[n+1],soln)," LOOPs Test ",

end;

# FUNCTIONS FOR SEQUENTIAL PROCESSING OF LATIN SQUARES


compare := function(e1,e2) # local function used to find position in list
if e1[1]<e2[1] then return true; else return false;fi;
end;

# firstTtab(dim) -- generates first Latin square of order dim in standard
# lexicographic order

firstTab := function(dim)
local n, c, r, tab2, complete, ulset, j, i, available, tryelem, direction;

	complete := [];
	available := [];
	n :=  dim;
  	tab2 :=  NullMat(n,n);
	complete := [1..n];
	
#  Moveback from end of table until a higher value can 
#  legally replace current. Place 0 in cells processed which
#  do not meet this condition.

	r := 1;  c:= 1;
	direction := "right";

while n>0 
do
	available := [];
	ulset := [];
	for j in [1..r] do AddSet(ulset, tab2[j][c]); od; 
	for j in [1..c] do AddSet(ulset, tab2[r][j]); od;		
	available := Difference(complete,ulset);
	tryelem := n + 1;

	while Size(available) > 0 
		do		   
		   tryelem := RepresentativeSmallest(available); 
		   if tryelem > tab2[r][c] 
			then tab2[r][c] := tryelem;
			break;
		   fi;		   
 		   Remove(available,1);		
		 od;
	if direction = "left" 
		then if tab2[r][c]<>tryelem 
			then tab2[r][c] := 0;
			else direction := "right"; 
		     fi;
	else 	     if tab2[r][c]<>tryelem
			then tab2[r][c] := 0;
			direction := "left";
		     fi;
	fi;

#  move to next cell
	if direction = "right"
		then if c = n then c:= 0; r := r + 1; fi;
		c := c + 1;
		if r = n+1 then return tab2; fi;
	else 
	 	if c = 1 then c:= n + 1; r := r - 1; fi;
		c := c - 1;
		if  r=0 then return []; fi;
	fi;	
od;

Print ("exhausted loop - program error ");
end;

# nextTab(tab) generates the Latin square which follows tab in standard 
# lexicographic order.

nextTab := function(tab)
local n, c, r, tab2, complete, ulset, j, i, available, tryelem, direction,count;
 


#  position at row n column n and process higher until digit can be 
#  increased

	complete := [];
	available := [];
	
	n :=  DimensionsMat(tab)[1];
  	tab2 :=  NullMat(n,n);

# duplicate tab in tab2

	for r in [1..n] do
		for c in [1..n] do
			tab2[r][c] := tab[r][c]; 
		od;
	od;

	complete := [1..n];
	
#  Moveback from end of table until a higher value can 
#  legally replace current. Place 0 in cells processed which
#  do not meet this condition.

	r := n;  c:= n;
	direction := "left";

while n > 0
do
		
	available := [];
	ulset := [];
	for j in [1..r] do AddSet(ulset, tab2[j][c]); od; 
	for j in [1..c] do AddSet(ulset, tab2[r][j]); od;		
	available := Difference(complete,ulset);
	tryelem := n + 1;


	while Size(available) > 0 
		do		   
		   tryelem := RepresentativeSmallest(available); 
		   if tryelem > tab2[r][c] 
			then tab2[r][c] := tryelem;
			break;
		   fi;		   
 		   Remove(available,1);		
		 od;
	if direction = "left" 
		then if tab2[r][c]<>tryelem 
			then tab2[r][c] := 0;
			else direction := "right"; 
		     fi;
	else 	     if tab2[r][c]<>tryelem
			then tab2[r][c] := 0;
			direction := "left";
		     fi;
	fi;

#  move to next cell
	if direction = "right"
		then if c = n then c:= 0; r := r + 1; fi;
		c := c + 1;
		if r = n+1 then return tab2; fi;
	else 
	 	if c = 1 then c:= n + 1; r := r - 1; fi;
		c := c - 1;
		if  r=0 then return []; fi;
	fi;	
od;
Print ("exhausted loop");
end;

# isoCounts(first,repets) Processes Cayley tables sequentially starting at first
# and for each spectrum counts the number of Cayley tables in a each isomorphic
# set. Terminates after repets tables have been processed or at the last
# table in the lexicographic sequence.

isoCounts := function(first,repets)
local tr,isos,t,iter,inx,p;
isos := [];

t :=  ShallowCopy(first);
AddSet(isos,[pSpectrum(t),t,1]);
for iter in [1..repets] do
   t := nextTab(t);
   if t = [] then break;fi;
   
   p := pSpectrum(t);
   inx:= PositionSorted(isos,[p,t,0],compare);   
      if inx>Length(isos) or isos[inx][1]<>p  then AddSet(isos,[p,t,1]);
      else
         while isos[inx][1]=p do
         if isomorphtab(t,isos[inx][2])<>fail then 
            isos[inx][3]:=isos[inx][3]+1;break;fi;
         inx:=inx+1; 
         if inx>Length(isos) or  isos[inx][1]<>p 
                     then AddSet(isos,[p,t,1]);break; fi;
         od; 
       fi;    
od;
return isos;
end;
